# Finding repeats: greedy LZ77 over a 32 KiB window, with hash chains.
#
# Every position is filed under the three bytes that start there. `heads`
# maps those three bytes, read as one number, to the latest position filed
# under them, and `links` holds, for each position in order, the position
# filed under the same three bytes before it (0 for none). Walking a chain
# from `heads` through `links` visits earlier occurrences nearest first, the
# order a match finder wants, since near matches cost fewer distance bits.
#
# A map is the only structure here that can be changed at a key, so `heads`
# is one; `links` only ever grows at the end, so it is a list.
import "../deflate"
import "std/bits"

type chains
  heads
  links

type found
  distance
  length

# How many earlier occurrences to try before settling. zlib's default level
# tries 128 with lazy matching; this greedy finder stops sooner.
chain_limit = 32

window = 32768

# One fixed-code block holding the whole input.
pub fn lz77_fixed src
  header = emit (emit empty_sink 1 1) 1 2
  body = scan src 1 (length src) header (chains {} [])
  finish (literal body 256)

fn scan src i n out ch
  return out if i > n
  return literal_run src i n out if i + 2 > n
  key = trigram src i
  first = ch.heads[key]
  best = longest src i n ch.links first (found 0 0) chain_limit
  filed = file ch key i (or_zero first)
  take src i n out filed best

fn take src i n out ch (found distance len)
  return scan src (i + 1) n (literal out (byte src i)) ch if len < 3
  more = file_range src (i + 1) (i + len - 1) n ch
  scan src (i + len) n (match_codes out len distance) more

# The last one or two bytes cannot start a match.
fn literal_run src i n out
  return out if i > n
  literal_run src (i + 1) n (literal out (byte src i))

fn trigram src i
  bits/shl (byte src i) 16 | bits/shl (byte src (i + 1)) 8 | byte src (i + 2)

fn byte src i
  deflate/byte_at src i

fn file (chains heads links) key i prior
  chains (put heads key i) (push links prior)

fn or_zero none
  0

fn or_zero p
  p

# File the positions a match covered, so later matches can start inside it.
fn file_range src from to n ch
  return ch if from > to or from + 2 > n
  key = trigram src from
  prior = or_zero ch.heads[key]
  file_range src (from + 1) to n (file ch key from prior)

# Walk the chain from `p`, keeping the longest match, until it runs out, leaves
# the window, or `tries` reaches zero.
fn longest _ _ _ _ none best _
  best

fn longest src i n links p best tries
  return best if tries == 0 or p == 0 or i - p > window
  limit = if (n - i + 1 < 258) (n - i + 1) 258
  len = match_length src p i 0 limit
  better = if (len > best.length) (found (i - p) len) best
  return better if len == limit
  longest src i n links (deflate/nth links p) better (tries - 1)

fn match_length src a b k limit
  return k if k == limit or src[a + k] != src[b + k]
  match_length src a b (k + 1) limit
